cmake_minimum_required(VERSION 3.16)
project(dubins_native LANGUAGES CXX)
add_library(dubins_native SHARED core.cpp terrain.cpp rrt_star.cpp bindings.cpp)
target_compile_features(dubins_native PUBLIC cxx_std_17)
target_compile_definitions(dubins_native PRIVATE DUBINS_BUILDING_LIBRARY INTERFACE DUBINS_USING_LIBRARY)
target_include_directories(dubins_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(dubins_native PRIVATE /O2 /fp:strict /W4)
else()
  target_compile_options(dubins_native PRIVATE -O3 -fno-fast-math -ffp-contract=off -Wall -Wextra)
endif()
set_target_properties(dubins_native PROPERTIES OUTPUT_NAME _native PREFIX "")
install(TARGETS dubins_native LIBRARY DESTINATION dubins_airplane RUNTIME DESTINATION dubins_airplane)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/examples/viewer.html" DEMO_HTML)
# Preserve newlines while splitting the embedded template into short literals.
string(REPLACE "\n" ")DubinsViewer\" \"\\n\"\nR\"DubinsViewer(" DEMO_HTML "${DEMO_HTML}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file(examples/viewer.hpp.in generated/viewer.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS examples/viewer.html)
add_library(dubins_demo OBJECT examples/scenarios.cpp examples/visualization.cpp)
target_include_directories(dubins_demo PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(dubins_demo PUBLIC dubins_native)
if(MSVC)
  target_compile_options(dubins_demo PRIVATE /utf-8)
endif()
foreach(name IN ITEMS basic altitude delivery obstacles)
  if(name STREQUAL "basic")
    set(executable example_cpp)
  else()
    set(executable example_${name}_cpp)
  endif()
  add_executable(${executable} example.cpp)
  target_compile_definitions(${executable} PRIVATE DUBINS_DEFAULT_SCENARIO="${name}")
  target_link_libraries(${executable} PRIVATE dubins_demo)
endforeach()

include(CTest)
if(BUILD_TESTING)
  foreach(name IN ITEMS basic altitude delivery obstacles)
    if(name STREQUAL "basic")
      set(executable example_cpp)
    else()
      set(executable example_${name}_cpp)
    endif()
    add_test(NAME cpp_${name} COMMAND ${executable} --output "${CMAKE_CURRENT_BINARY_DIR}/smoke/${name}")
  endforeach()
endif()
